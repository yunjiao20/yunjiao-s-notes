#include <stdio.h>
#include <stdlib.h>
#include <assert.h>

int main()
{
	/*
	关掉标准输出的缓冲。不加这句的话，printf
	的输出会攒在缓冲区里，在 gdb 里单步或管
	道重定向时看不到实时输出，影响调试。
	*/
	setbuf(stdout, NULL);

	printf("This file demonstrates a simple double-free attack with fastbins.\n");



	/* 填满tcache

	tcache 是 glibc 2.26 引入的一个每线程独立的小缓存，本质是一个数组 + 单链表，
	free 掉的 chunk 会先放在tcache，下次 malloc 相同大小直接从 tcache 拿，不用走
	复杂的 bin 逻辑，速度快很多。

	glibc中，只要 tcache 没满，free 一律优先进 tcache

        tcache 的结构是一个数组，数组的每个位置（叫 bin / 索引）对应一个固定的 chunk 大小档位：
		tcache 索引 0  → 专门存 0x20 大小的 chunk
		tcache 索引 1  → 专门存 0x30 大小的 chunk
		tcache 索引 2  → 专门存 0x40 大小的 chunk
		tcache 索引 3  → 专门存 0x50 大小的 chunk
		...
		tcache 索引 63 → 专门存 0x410 大小的 chunk
	共 64 个 bin
	• 同一个 bin 里，所有 chunk 大小完全一样（都是这一档的大小）
	• 不同 bin 之间，大小不一样（0x20、0x30、0x40... 逐档递增）
	你的 chunk 是多大，free 后就进对应的那一档

	tcache 的每一个 bin 最多放 7 个chunk
	*/
	printf("Fill up tcache first.\n");
	void *ptrs[8];
	for (int i=0; i<8; i++) {
		/* 循环8次，在堆上创建 8 个 0x20 字节的块

		malloc(8) 请求8字节，header需要16字节，8+16=24，chunk要求16字节（0x10）
		对齐，这里对齐到 0x20 ，在堆上获取一个大小为 0x20 的块
		*/
		ptrs[i] = malloc(8);
	}
	for (int i=0; i<7; i++) {
		/* free 其中 7 个块

		前面讲过，tcache 的每个 bin 只能挂 7 个chunk
		这里每个 free 的 chunk 大小均为 0x20，free 后都进 0x20 的 bin 。
		- 每次 free，chunk 进 tcache 的 0x20 索引，计数 +1
		- 第 7 个 free 完，计数 = 7，这个 bin 满了
		所以后面 free 的 0x20 大小的 chunk 不能放在 tcache 中（相应的bin
		满了），后面的 free 无处可去，只能进 fastbin

		所以只 free 了七个 chunk，第八个 chunk 留着不释放*/
		free(ptrs[i]);
	}



	/* 使用 calloc 分配三块内存

	这里使用 calloc 而不用 malloc 是因为：
		malloc  分配非全零内存  会先查 tcache，从 tcache 取
		calloc  承诺全零内存    会跳过 tcache
	calloc 承诺返回全零的内存，而 tcache 里的 chunk 是之前 free 掉的、可能残留
	脏数据，glibc 为了「保证清零」，让 calloc 直接不碰 tcache，从 top chunk（堆
	顶的空白区）切新块

	这是为了保证 `tcache 已满` 的状态，保证后面的 free 进 fastbin
	如果这里用 malloc，会先把 tcache 里那 7 个 chunk 吐出来，tcache 没有满，后面
	的 free 就会再进 tcache ，和我们学习 fastbin 的目的不符
	*/
	printf("Allocating 3 buffers.\n");
	int *a = calloc(1, 8);
	int *b = calloc(1, 8);
	int *c = calloc(1, 8);

	printf("1st calloc(1, 8): %p\n", a);
	printf("2nd calloc(1, 8): %p\n", b);
	printf("3rd calloc(1, 8): %p\n", c);

	/* 第一次 free
	• 查 tcache：满了（7/7）→ 进不去
	• 于是进 fastbin（0x20 属于 fastbin 范围）
	• fastbin 是单链表 LIFO（后进先出），新 free 的放最前面

	于是 fastbin 的结构变成：
		fastbin[0x20]:  a  （a 是链表头）
	*/
	printf("Freeing the first one...\n");
	free(a);


	/* glibc 会做 fastbin double free 检测：检查要 free 的 chunk 是不是当前链表头？
	如果是，会报错（如 free(a) -> free(a) 报错）。所以如果你解开下面的 free(a) 的注
	释，编译时会报错
	但是fastbin只检查链表头的审查很弱，free(a) -> free(b) -> free(a) 就可以绕过检测*/
	printf("If we free %p again, things will crash because %p is at the top of the free list.\n", a, a);
	// free(a);


	/* 第二次 free

	glibc 检查 b 不是链表头，同理进 fastbin，b 变成新的链表头
		fastbin[0x20]:  b -> a
	*/
	printf("So, instead, we'll free %p.\n", b);
	free(b);


	/* 再次free a，double free

	glibc 做 fastbin double free 检测：要 free 的 a 是不是当前链表头？而当前链表头
	是 b，不是 a 。检测通过，a 再次被塞进链表头
		fastbin[0x20]:  a -> b -> a
	*/
	printf("Now, we can free %p again, since it's not the head of the free list.\n", a);
	free(a);


	/* 三次 calloc，把 fastbin 中的 a→b→a 环消费掉*/
	printf("Now the free list has [ %p, %p, %p ]. If we malloc 3 times, we'll get %p twice!\n", a, b, a, a);
	a = calloc(1, 8);        // a , 剩余链表 b -> a
	b = calloc(1, 8);        // b , 剩余链表 a
	c = calloc(1, 8);        // a , 剩余链表 [空]，获取了两次 a
	printf("1st calloc(1, 8): %p\n", a);
	printf("2nd calloc(1, 8): %p\n", b);
	printf("3rd calloc(1, 8): %p\n", c);


	/*
	assert 函数在条件不成立时会立即终止程序并报告错误位置
	这里我们检查 a 和 c 是否指向同一块内存，assert(a == c) 成立

	否则终止程序，证明在当前libc版本下此例子不可用
	*/
	assert(a == c);
}






/* fastbin

（来源对AI信息的整理提纯）

fastbin 是 glibc 里专门给「小块内存」做快速分配/释放的一批链表，比 tcache 早很多（远古
版本就有）。
程序里绝大多数 malloc 都是很小的、频繁分配又频繁释放的（比如一个 16 字节的字符串节点）。
为这些小 chunk 走完整的 bin 逻辑太慢，所以搞了个「快速通道」——fastbin。

和 tcache 一样，fastbin 也是按档位分箱，但范围和档位不同：

fastbin 数组：
  fastbin[0] → 0x20 档
  fastbin[1] → 0x30 档
  fastbin[2] → 0x40 档
  fastbin[3] → 0x50 档
  fastbin[4] → 0x60 档
  fastbin[5] → 0x70 档
  fastbin[6] → 0x80 档
超过 0x80 的 chunk free 之后，就不走 fastbin了，进 unsorted bin 或按大小进 small/large bin。

特点 1：单链表，只用 fd 字段
一个被 free 的 chunk，它头部那 16 字节后面（用户数据区的前 8 字节）会被拿来存 fd，指向
链表里的下一个 chunk。

free 状态的 fastbin chunk：
+------------------+
| prev_size        |
+------------------+
| size             |
+------------------+
| fd  → 下一个chunk|   ← 用户数据区前 8 字节被复用
+------------------+

特点 2：free 后不清除数据、不清 PREV_INUSE 位
这是 fastbin 最特殊的地方。普通 chunk free 之后会合并（consolidate）、会更新标志位，
但 fastbin 为了「快」，free 的时候不合并、不清除下一个 chunk 的 PREV_INUSE 位。
• 好处：快
• 坏处：chunk 上残留的旧数据、旧指针还在，给利用留了空间

特点 3：LIFO，新 free 的放链表头
free 顺序是先 a 后 b，链表就是 b → a（b 在头）。malloc 时从链表头取，所以先取回 b。

四、double free 检测机制（回到那个例子）
glibc 对 fastbin 有一个简单的 double free 检测——注意，它很弱：
    只检查「当前要 free 的 chunk 是不是链表头」
注意它只比对了「链表头」这一块，没扫描整个链表。 所以：
• free(a); free(a) → 第二次 free 时链表头是 a，等于 p → 检测触发，崩
• free(a); free(b); free(a) → 第二次 free(a) 时链表头是 b，不等于 a → 检测不触发，绕过成功
这就是为什么 fastbin_dup 里必须「夹一个 free(b)」。

五、fastbin 和 tcache 的关系（2.26 之后）
现代 glibc 里必须搞懂的一层：
1. tcache 优先级更高：free 一个 chunk，只要 tcache 没满，优先进 tcache，根本不轮到 fastbin。
2. tcache 容量小：每档 7 个，满了就装不下。
3. tcache 被「跳过」的入口：前面说的 calloc 会跳过 tcache，直接操作 fastbin。
所以现代环境里想玩 fastbin，几乎都要先做一件事：填满 tcache（7个），把 free 逼进 fastbin。
这正是 fastbin_dup.c 开头那 7 次 free 的意义。



一句话总结：
fastbin 是「小 chunk 的快速单链表通道」，LIFO、只用 fd、free 不合并，double free 检测只
查链表头——这个「只查头」的弱点，就是 fastbin_dup 的立足点。而在 2.26 之后，想碰到 fastbin，
得先填满 tcache。

• fastbin 是 glibc bin 体系的一部分。它和 small bin、large bin、unsorted bin 一起，被
  malloc/free 的核心分配器（_int_malloc/_int_free）统一管理。它要维护全局的一致性、要
  走那套复杂的 bin 逻辑。

• tcache 是「独立于 bin 体系之外」的一层旁路缓存。它在分配器最前面拦一道，free 先进
  tcache、malloc 先查 tcache，根本不进入核心 bin 逻辑。

一句话：fastbin 是「主路里的一条快速车道」，tcache 是「主路之外单独开的一条捷径」。

  |     维度    |              fastbin              |              tcache              |
  |:----------- |:--------------------------------- |:-------------------------------- |
  |引入版本     |远古                               |2.26                              |
  |大小范围     |0x20 ~ 0x80                        |0x20 ~ 0x410                      |
  |档位数       |7                                  |64                                |
  |每档上限     |无（理论不限）                     |7 个                              |
  |free 优先级  |低（tcache 满了才轮到）            |高（先走它）                      |
  |顺序         |LIFO                               |LIFO                              |
  |归属         |bin 体系内                         |bin 体系外的旁路缓存              |
  |数据结构     |单链表（fd）                       |单链表（fd） + 额外一个 count 计数|
  |大小范围     |0x20  0x80（7 档）                 |0x20  0x410（64 档）              |
  |每档上限     |无                                 |7 个（有计数限制）                |
  |线程         |全局 arena，要加锁                 |每线程独立，基本无锁              |
  |free 时检查  |有 double free 检测（弱）等        |早期几乎零检查（2.29 后才补）     |
  |malloc 时处理|走 _int_malloc 完整流程，有各种边界|拿到就返回，几乎不做处理          |
  |chunk 标志位 |维护 PREV_INUSE 等                 |free 后基本不动标志位             |

### 为什么有了 fastbin 还要 tcache？
一句话：为了多线程性能，把「快」推到极致。
fastbin 虽然叫「fast」，但它有两个开销省不掉：
1. 锁。fastbin 挂在全局 arena 上，多个线程同时 malloc/free 要抢锁。高并发下锁竞争是性能
   杀手。
2. 一致性维护。fastbin 再快，也还是要走 _int_free 那套：检查、维护标志位、处理边界，每一
   步都是 CPU 指令。
而现代程序（尤其浏览器、大型服务端）是重度多线程 + 疯狂小对象分配，fastbin 的这两个开销
被无限放大。

tcache 的解法极其简单粗暴：
• 每个线程一个独立的 tcache，自己的缓存自己用，基本不用加锁 → 锁竞争消失
• free 进去几乎什么都不检查，malloc 拿出来几乎什么都不处理 → 一致性维护省掉
• 用一个 count 计数限制每档 7 个，防止无限堆积
这套设计的来源其实是 2015 年 Google 在 glibc 社区提的一个 tcache 补丁（参考了 jemalloc 等
现代分配器的 per-thread cache 思想），后来被合入 2.26。本质就是「用一点内存换巨大的多线程
性能提升」。

### 这个区别对堆利用意味着什么（重点）
作为学堆利用的人，你要记住的是：
1. tcache 优先级永远高于 fastbin。free 一个 chunk，只要 tcache 没满就先进 tcache，fastbin
   根本轮不到。所以现代环境玩 fastbin 必先填满 tcache——这正是 fastbin_dup.c 开头那 7 次
   free 的由来。
2. tcache 的「零检查」是双刃剑。它为了快省略的检查，正是攻击者的机会。2.262.28 的 tcache
   几乎无防护，tcache poisoning 极其好用；后来 2.29 加了 double free 检测、2.32 加了
   safe-linking 加密 fd，就是官方在慢慢「补上这些省掉的检查」。
3. 理解「旁路」这个概念，后面学 calloc 跳过 tcache、_int_malloc 里的 tcache 提取逻辑（
   tcache_stashing_unlink_attack 就是打这里）时，你会豁然开朗。

一句话总结：
fastbin 是 bin 主路里的小块快速通道，要锁、要维护一致性；tcache 是主路之外每线程独立的旁
路缓存，几乎零检查零锁。引入 tcache 是为了用「省掉锁和检查」来换多线程性能，而省掉的那些
检查，反过来成了堆利用的主战场。
*/
